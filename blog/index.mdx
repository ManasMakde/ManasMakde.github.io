import { HTMLSkeleton, Header, SearchBar, NavBar, PaginationBar, Footer } from "@/components.jsx";


export function ArticlePreview({ id, imgSrc, title, date, description, link, isLoading = false, loadingClass = "loading" }) {
    const removeLoadingFunction = isLoading ? `this.parentElement.classList.remove('${loadingClass}')` : "";
    return (<div id={id} className={`article-preview ${isLoading ? loadingClass : ""}`}>
        <a href={link} className="article-preview-image-link">
            <img className={isLoading ? loadingClass : ""} src={imgSrc} onLoad={removeLoadingFunction} onError={removeLoadingFunction} alt="thumbnail" fetchPriority="high" />
        </a>

        <div className="article-preview-body">
            <a href={link} className="article-preview-title-link">
                <h2 className="article-preview-title">{title}</h2>
            </a>
            <span className="article-preview-date">{date}</span>
            <p className="article-preview-description">{description}</p>
        </div>
    </div>)
}
