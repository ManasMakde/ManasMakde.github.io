Scale via shader in godot

Posted: 26 November 2023
shader_type canvas_item;

uniform vec2 scale = vec2(2.0, 2.0); // x and y scale

void vertex() {
    VERTEX.x *= scale.x;
    VERTEX.y *= scale.y;
}
