export function SnippetPreview({ imgSrc, text, link, tags = [], isLoading = false, loadingClass = "is-loading" }) {

    const removeLoadingFunction = isLoading ? `this.parentElement.classList.remove('${loadingClass}')` : ""

    return (<div className="snippet-preview">
        <a href={link} className={`snippet-preview-thumbnail ${isLoading ? loadingClass : ""}`} tabindex="-1">
            <img src={imgSrc} onerror={removeLoadingFunction} onLoad={removeLoadingFunction} alt="thumbnail" fetchpriority="high" />
        </a>
        <a href={link} className={`snippet-preview-title ${isLoading ? loadingClass : ""}`} title={text}>
            <div>{text}</div>
        </a>
        <div className={`preview-tag-container ${isLoading ? loadingClass : ""}`}>
            <Tags tags={tags} assignHref={false} />
        </div>
    </div>)
}
